decl
    int A[10][10], B[10][10], C[10][10];
    int n;
    int i, j, k;
enddecl
begin
    read(n);

    i = 0;
    while(i < n) do
        j = 0;
        while(j < n) do
            read(A[i][j]);
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

    i = 0;
    while(i < n) do
        j = 0;
        while(j < n) do
            read(B[i][j]);
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

    i = 0;
    while(i < n) do
        j = 0;
        while(j < n) do
            C[i][j] = 0;
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

    i = 0;
    while(i < n) do
        j = 0;
        while(j < n) do
            k = 0;
            while(k < n) do
                C[i][j] = C[i][j] + (A[i][k] * B[k][j]);
                k = k + 1;
            endwhile;
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

    i = 0;
    while(i < n) do
        j = 0;
        while(j < n) do
            write(C[i][j]);
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;
end;
